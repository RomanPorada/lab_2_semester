# FIXME: Ця функція падає, якщо передано None
def process_user_data(user_data):
    if not user_data:
        # BUG: Це призводить до падіння при обробці пустих словників
        return None
    
    # HACK: Швидке рішення для отримання імені, працює не завжди коректно
    name = user_data.get("name", "Unknown")
    
    # NOTE: Ми припускаємо, що всі користувачі мають унікальні ID
    user_id = user_data.get("id", 0)
    
    print(f"Processing user: {name} (ID: {user_id})")
    
    # OPTIMIZE: Зробити кешування результатів для великих обсягів даних
    result = do_expensive_calculation(user_id)
    
    return result

# REVIEW: Чи правильно ця функція обробляє крайні випадки?
def do_expensive_calculation(user_id):
    total = 0
    for i in range(1, 10000):
        total += i * user_id
    return total
